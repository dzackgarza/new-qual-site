$$
( \mathrm { a } ) \sum _ { n = 1 } ^ { \infty } [ 1 - \operatorname { t a n h } ( n ) ] , \quad \quad ( \mathrm { b } ) \sum _ { n = 1 } ^ { \infty } \left( { \frac { \pi } { 2 } } - \arctan ( n ) \right) .
$$
